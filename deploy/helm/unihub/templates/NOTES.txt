Release Name: {{ .Release.Name }}
Namespace:    {{ .Release.Namespace }}
Environment:  {{ .Values.global.environment }}

1. Check Pod status:
   kubectl get pods -n {{ .Release.Namespace }} -l app.kubernetes.io/part-of=unihub

2. Access the Application:
{{- if .Values.ingress.enabled }}
   Ingress is ENABLED. You can access the services via:
   {{- range .Values.ingress.hosts }}
   {{- if .host }}
   - Host: http://{{ .host }}
   {{- else }}
   - Host: (IP-based default host)
   {{- end }}
   {{- end }}
{{- else if eq .Values.api.service.type "NodePort" }}
   Backend API is exposed via NodePort:
   export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services unihub-api-service)
   export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo "API URL: http://$NODE_IP:$NODE_PORT"
   echo "Health Check: http://$NODE_IP:$NODE_PORT/health"
{{- else }}
   Port-forward to Backend API:
   kubectl port-forward -n {{ .Release.Namespace }} svc/unihub-api-service 8080:80
   Health Check: http://localhost:8080/health

   Port-forward to Frontend Web:
   kubectl port-forward -n {{ .Release.Namespace }} svc/unihub-web-service 3000:80
   Web UI: http://localhost:3000
{{- end }}

3. Metrics & Monitoring:
   Prometheus scraping endpoint is configured at: /metrics on unihub-api-service.
